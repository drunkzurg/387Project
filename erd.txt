// ========== USERS ==========

Table users {
  user_id integer [primary key]
  name varchar
  email varchar
  password_hash varchar
  role varchar                // sys_admin, owner, employee, hr
  pending_approval boolean
}

// ========== DEPARTMENTS ==========

Table departments {
  department_id integer [primary key]
  name varchar
  department_type varchar     // play_area, gift_shop, customer_support
  entrance_fee_tickets integer
  operating_status varchar    // active, out_of_order, inactive
  description varchar
  created_at datetime
  updated_at datetime
}

// ========== EMPLOYEES ==========

Table employees {
  employee_id integer [primary key]
  user_id integer [not null]
  name varchar
  department_id integer
  hourly_wage float
  status varchar              // active, transferred, terminated
}

// ========== EMPLOYEE SHIFTS ==========

Table employee_shifts {
  shift_id integer [primary key]
  employee_id integer [not null]
  start_time datetime
  end_time datetime
}

// ========== EMPLOYEE HOURS ==========

Table employee_hours {
  log_id integer [primary key]
  employee_id integer [not null]
  date date
  hours_worked float
}

// ========== EMPLOYEE TRANSFERS ==========

Table employee_transfers {
  transfer_id integer [primary key]
  employee_id integer [not null]
  from_department integer
  to_department integer
  date date
  handled_by_user_id integer
}

// ========== ATTENDEES ==========

Table attendees {
  attendee_id integer [primary key]
  name varchar
  email varchar
  membership_code varchar
  phone varchar
  is_member boolean
  verified_by_employee_id integer
  verified_at datetime
  created_at datetime
}

// ========== ATTENDEE SESSIONS ==========

Table attendee_sessions {
  session_id integer [primary key]
  attendee_id integer
  department_id integer [not null]
  employee_id integer [not null]
  display_name varchar
  admission_mode varchar      // walk_in, member_wallet, manual_override
  entrance_fee_tickets integer
  payout_tickets integer
  notes varchar
  opened_at datetime
  closed_at datetime
}

// ========== GIFT SHOP ITEMS ==========

Table gift_shop_items {
  gift_shop_item_id integer [primary key]
  name varchar
  ticket_price integer
  cost_price float
  stock integer
  status varchar              // active, inactive
  category varchar
  description varchar
  created_at datetime
  updated_at datetime
}

// ========== TICKET ACCOUNTS ==========

Table ticket_accounts {
  ticket_account_id integer [primary key]
  account_code varchar
  account_kind varchar        // gift_shop_budget, gift_shop_revenue, gift_shop_investment, department_reserve, department_generated, member_wallet, session_wallet
  department_id integer
  attendee_id integer
  attendee_session_id integer
  balance integer
  created_at datetime
  updated_at datetime
}

// ========== TICKET TRANSACTIONS ==========

Table ticket_transactions {
  ticket_transaction_id integer [primary key]
  transaction_type varchar    // department_admission, department_payout, gift_shop_redemption, owner_allocation, owner_generated_transfer, owner_investment, member_claim_transfer, manual_override
  source_account_id integer
  destination_account_id integer
  amount integer
  department_id integer
  attendee_id integer
  attendee_session_id integer
  employee_id integer
  gift_shop_item_id integer
  created_by_user_id integer
  note varchar
  created_at datetime
}

// ========== GIFT SHOP REDEMPTIONS ==========

Table gift_shop_redemptions {
  redemption_id integer [primary key]
  gift_shop_item_id integer [not null]
  department_id integer [not null]
  employee_id integer [not null]
  attendee_id integer
  attendee_session_id integer
  source_account_id integer [not null]
  quantity integer
  total_tickets integer
  notes varchar
  redeemed_at datetime
}

// ========== REPORTS ==========

Table revenue_reports {
  report_id integer [primary key]
  total_ticket_admissions integer
  total_ticket_payouts integer
  total_gift_shop_redemptions integer
  total_owner_investment integer
  active_attendees integer
  start_date date
  end_date date
  generated_by integer
  created_at datetime
}

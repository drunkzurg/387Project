// Arcade Management System — physical data model (dbdiagram.io / Vertabelo friendly).
// Copy of the root erd.txt for convenience (keep in sync).
// Source of truth for DDL: sql/migrations/000_run_all.sql
//
// ----- Operating budget (tickets, not dollars) -----
// gift_shop_budget is the shared pool: owner investment, admissions, and transfers IN;
// play-area payouts and gift-shop inventory procurement OUT. gift_shop_inventory_spend
// tracks cumulative procurement into an audit/holding account. See TicketService.

// ========== USERS ==========

Table users {
  user_id integer [primary key]
  name varchar
  email varchar
  password_hash varchar
  role varchar                // sys_admin, owner, employee, hr (ENUM in DB)
  pending_approval boolean
}

// ========== DEPARTMENTS ==========

Table departments {
  department_id integer [primary key]
  name varchar
  department_type varchar     // play_area, gift_shop, customer_support
  entrance_fee_tickets integer
  capacity integer
  operating_status varchar    // active, out_of_order, inactive
  description varchar
  created_at datetime
  updated_at datetime
}

// ========== EMPLOYEES ==========

Table employees {
  employee_id integer [primary key]
  user_id integer [not null]
  name varchar
  department_id integer
  hourly_wage float           // DECIMAL in DB
  status varchar              // active, transferred, terminated
}

// ========== EMPLOYEE SHIFTS ==========

Table employee_shifts {
  shift_id integer [primary key]
  employee_id integer [not null]
  start_time datetime
  end_time datetime
  entry_type varchar          // live, manual — clock-in vs manual backfill
}

// ========== EMPLOYEE SICK REQUESTS ==========

Table employee_sick_requests {
  sick_request_id integer [primary key]
  employee_id integer [not null]
  request_date date
  status varchar              // waiting, approved, denied
  notes varchar
  requested_at datetime
  reviewed_by_user_id integer
  reviewed_at datetime
  review_notes varchar
}

// ========== HR ACTION LOGS ==========

Table hr_action_logs {
  hr_action_log_id integer [primary key]
  action_type varchar
  employee_id integer
  handled_by_user_id integer [not null]
  details varchar
  created_at datetime
}

// ========== ATTENDEES ==========

Table attendees {
  attendee_id integer [primary key]
  name varchar
  email varchar
  membership_code varchar
  phone varchar
  is_member boolean
  verified_by_employee_id integer
  verified_at datetime
}

// ========== ATTENDEE SESSIONS ==========

Table attendee_sessions {
  session_id integer [primary key]
  attendee_id integer
  department_id integer [not null]
  employee_id integer [not null]
  display_name varchar
  admission_mode varchar      // walk_in, member_wallet, manual_override
  entrance_fee_tickets integer
  payout_tickets integer
  notes varchar
  opened_at datetime
  closed_at datetime
}

// ========== GIFT SHOP ITEMS ==========

Table gift_shop_items {
  gift_shop_item_id integer [primary key]
  name varchar
  ticket_price integer
  cost_price float            // DECIMAL — tickets per unit for stocking
  stock integer
  status varchar              // active, inactive
  category varchar
  description varchar
  created_at datetime
  updated_at datetime
}

// ========== TICKET ACCOUNTS ==========

Table ticket_accounts {
  ticket_account_id integer [primary key]
  account_code varchar
  account_kind varchar        // gift_shop_budget, gift_shop_revenue, gift_shop_investment, gift_shop_inventory_spend, department_reserve, department_generated, member_wallet, session_wallet
  department_id integer
  attendee_id integer
  attendee_session_id integer
  balance bigint              // BIGINT UNSIGNED in DB
  created_at datetime
  updated_at datetime
}

// ========== TICKET TRANSACTIONS ==========

Table ticket_transactions {
  ticket_transaction_id integer [primary key]
  transaction_type varchar    // department_admission, department_payout, gift_shop_redemption, gift_shop_inventory_procurement, gift_shop_inventory_credit, owner_allocation, owner_generated_transfer, owner_investment, member_claim_transfer, manual_override
  source_account_id integer
  destination_account_id integer
  amount bigint               // BIGINT UNSIGNED in DB
  department_id integer
  attendee_id integer
  attendee_session_id integer
  employee_id integer
  gift_shop_item_id integer
  created_by_user_id integer
  note varchar
  created_at datetime
}

// ========== GIFT SHOP REDEMPTIONS ==========

Table gift_shop_redemptions {
  redemption_id integer [primary key]
  gift_shop_item_id integer [not null]
  department_id integer [not null]
  employee_id integer [not null]
  attendee_id integer
  attendee_session_id integer
  source_account_id integer [not null]
  quantity integer
  total_tickets bigint        // BIGINT UNSIGNED in DB
  notes varchar
  redeemed_at datetime
}
